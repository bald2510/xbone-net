# Research-question experiment switches for tools/run_all.py.
#
# The selectors below are intentionally grouped by research question. Duplicate
# configs (especially the canonical XBone-Net reference) are de-duplicated by
# run_all.py before any subprocess is launched.
#
# Preview only:
#   python tools/run_all.py --dry-run
# List group membership:
#   python tools/run_all.py --list-configs

[enabled]

# RQ1 — controlled modality comparison on BTXRD:
# full multimodal, image-only, text-only, and cross-class shuffled text.
+ group:rq1_btxrd_modality

# RQ2 — component families. Existing leave-one-out config names and result
# paths are unchanged; this group only reorganizes how they are selected.
+ group:rq2_all_components

# RQ3 — reuse the three existing CTCH adaptation levels: frozen BiomedCLIP
# encoders, XBone-Net with LoRA, and XBone-Net with full fine-tuning. Historical
# no-FT metrics are reused without checkpoints; LoRA/full-FT training summaries
# are merged into the result table whenever available.
+ group:rq3_lora_vs_full_ft

# RQ4 — reuse existing XBone-Net, BioMedCLIP zero-shot, phase2-only, concat and
# linear-head classification results. The post-hoc runner reuses existing OOD
# archives for four trained models and only exports the deterministic zero-shot
# BioMedCLIP representation once at seed 42.
+ group:rq4_ood_representation_baselines

[disabled]

# Keep unrelated baseline and few-shot suites out of the default queue. They
# remain selectable explicitly with --ignore-experiment-file --group <name>.
