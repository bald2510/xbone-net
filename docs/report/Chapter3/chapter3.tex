\chapter{Phương pháp đề xuất}
\label{Chapter3}

\section{Tổng quan phương pháp đề xuất}
\label{sec:final-pipeline}

Nhóm nghiên cứu đề xuất XBone-Net cho bài toán phân loại đa lớp ảnh X-quang xương có kết hợp thông tin lâm sàng. Hệ thống được xây dựng theo kiến trúc hai nhánh. Nhánh ảnh khai thác đồng thời cấu trúc giải phẫu toàn cục và các vùng cục bộ giàu chi tiết, trong khi nhánh văn bản biểu diễn bệnh sử và lý do vào viện. Hai nguồn thông tin được ánh xạ vào cùng một không gian đặc trưng, trao đổi ngữ cảnh bằng cơ chế chú ý chéo hai chiều, sau đó được phân lớp bằng tâm cụm thực nghiệm.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{images/chapter3/tongquat.png}
    \caption{Sơ đồ tổng quan phương pháp XBone-Net đề xuất}
    \label{fig:ch3_overview}
\end{figure}

Như minh họa trong Hình \ref{fig:ch3_overview}, XBone-Net kế thừa bộ mã hóa thị giác và bộ mã hóa văn bản của BiomedCLIP \cite{BiomedCLIP}. Các tham số nền được giữ cố định để bảo toàn tri thức y sinh đã học trước, trong khi các ma trận thích nghi hạng thấp LoRA \cite{Hu2022LoRA} được huấn luyện để thích nghi trên dữ liệu đích. Sau bước mã hóa, mô-đun dung hợp kết hợp đặc trưng toàn cục và cục bộ của hai phương thức. Biểu diễn tổng hợp cuối cùng được so sánh với các nguyên mẫu lớp để đưa ra dự đoán. Định danh mô hình nền tảng, kích thước biểu diễn, vị trí chèn LoRA và cấu hình tái lập đầy đủ được trình bày tại Phụ lục~\ref{AppendixModelDetails}.

Mỗi mẫu dữ liệu được ký hiệu bởi bộ ba gồm ảnh X-quang, thông tin lâm sàng và nhãn lớp. Tập huấn luyện có dạng:

\begin{equation}
\mathcal{D}_{\mathrm{train}}
=
\left\{
\left(x_i^{\mathrm{img}},x_i^{\mathrm{txt}},y_i\right)
\right\}_{i=1}^{N},
\qquad
y_i\in\mathcal{Y}.
\label{eq:ch3_dataset}
\end{equation}

Trong đó, $x_i^{\mathrm{img}}\in\mathbb{R}^{H_i\times W_i\times C_{\mathrm{in}}}$ là ảnh X-quang của mẫu thứ $i$. Các ký hiệu $H_i$, $W_i$ và $C_{\mathrm{in}}$ lần lượt là chiều cao, chiều rộng và số kênh ảnh. Thông tin lâm sàng được ký hiệu $x_i^{\mathrm{txt}}=\left(w_{i,1},\ldots,w_{i,L_i}\right)$, với $L_i$ là số đơn vị biểu dmạng niễn trước khi cắt hoặc đệm. Nhãn $y_i$ thuộc tập chỉ số lớp $\mathcal{Y}=\{1,\ldots,K\}$, còn $N$ là số mẫu huấn luyện.

Quá trình xây dựng mô hình được chia thành hai giai đoạn ngoại tuyến. Giai đoạn thứ nhất thích nghi hai nhánh mã hóa và căn chỉnh không gian ảnh-văn bản bằng hàm mất mát khớp ngữ nghĩa. Giai đoạn thứ hai học cơ chế dung hợp và ranh giới phân lớp trong không gian nguyên mẫu. Khi suy luận trực tuyến, toàn bộ tham số được cố định, mô hình chỉ thực hiện tính toán truyền thuận cho một ca mới.

Để làm rõ cách hệ thống chuyển đổi từng nguồn dữ liệu thành biểu diễn có thể sử dụng cho quá trình dung hợp, các phần tiếp theo lần lượt trình bày nhánh biểu diễn ảnh, nhánh biểu diễn văn bản và cơ chế thích nghi hai bộ mã hóa trước khi đi đến mô-đun dung hợp và phân lớp.

\section{Nhánh mã hóa hình ảnh}

Mô hình kế thừa bộ mã hóa của BiomedCLIP \cite{BiomedCLIP}, được xây dựng theo kiến trúc Vision Transformer \cite{Dosovitskiy2021ViT} ViT-B/16. Hình ảnh y tế đầu vào $v$ ban đầu được chuẩn hóa về độ phân giải tiêu chuẩn $224 \times 224$ pixels với $3$ kênh màu (RGB). Quá trình biến đổi dữ liệu để trích xuất đặc trưng qua bộ mã hóa Vision Transformer ($\text{ViT-B/16}$) diễn ra tuần tự như sau:

\begin{enumerate}
    \item \textbf{Phân mảnh ảnh (Patch Extraction):} Hình ảnh $v \in \mathbb{R}^{224 \times 224 \times 3}$ được phân rã thành chuỗi $N$ mảnh ảnh nhỏ không chồng lấn (patches) có kích thước cố định là $16 \times 16$ pixels. Số lượng mảnh ảnh thu được, đồng thời là độ dài chuỗi đầu vào của mạng Transformer, được xác định bằng công thức:
    \begin{equation}
    N = \frac{224 \times 224}{16^2} = 196
    \end{equation}
    
    \item \textbf{Nhúng tuyến tính và chèn token $[CLS]$:} Các mảnh ảnh được trải phẳng và chiếu tuyến tính qua ma trận trọng số $E \in \mathbb{R}^{(16^2 \cdot 3) \times D}$ vào không gian ẩn có số chiều ẩn $D = 768$ để tạo thành các \textit{patch embeddings}. Để tổng hợp thông tin ngữ nghĩa toàn cục của bức ảnh, một token phân loại học được đặc biệt - ký hiệu là $v_{\text{CLS}} \in \mathbb{R}^{1 \times D}$ - được chèn vào ngay đầu chuỗi đặc trưng. Đồng thời, một ma trận nhúng vị trí một chiều học được $E_{\text{pos}} \in \mathbb{R}^{(N+1) \times D}$ được cộng trực tiếp vào chuỗi nhằm bảo toàn thông tin cấu trúc không gian:
    \begin{equation}
    z_0 = [v_{\text{CLS}}; v_p^1 E; v_p^2 E; \dots; v_p^N E] + E_{\text{pos}}
    \end{equation}
    
    \item \textbf{Xử lý qua Transformer Encoder:} Chuỗi biểu diễn tích hợp $z_0$ được truyền qua chồng gồm $L = 12$ tầng Transformer Encoder. Tại mỗi tầng, các cơ chế tự chú ý đa đầu (Multi-Head Self-Attention) và mạng truyền thẳng (MLP) liên tục cập nhật trạng thái của toàn bộ chuỗi thông qua chuẩn hóa trước và kết nối tắt residual.
    
    \item \textbf{Trích xuất đầu ra của bộ mã hóa:} Tại tầng mã hóa cuối cùng ($L=12$), đặc trưng biểu diễn toàn cầu của hình ảnh (ký hiệu là $y_v$) được trích xuất bằng cách lấy trạng thái ẩn của token phân loại ở vị trí đầu tiên của chuỗi ($z_L^0$) đi qua một lớp chuẩn hóa LayerNorm cuối cùng:
    \begin{equation}
    y_v = \text{LN}(z_L^0) \in \mathbb{R}^D
    \end{equation}
\end{enumerate}

\section{Nhánh biểu diễn văn bản}
\label{sec:ch3_text_branch}
Bổ sung cho các bằng chứng hình thái được trích xuất từ ảnh X-quang, nhánh biểu diễn văn bản có nhiệm vụ mã hóa thông tin lâm sàng $t$ thành các đặc trưng ngữ cảnh theo từng đơn vị biểu diễn và một vectơ đại diện cho toàn bộ chuỗi. Trong khóa luận, nhánh này được kế thừa từ bộ mã hóa văn bản của BiomedCLIP \cite{BiomedCLIP}, với thành phần chính là PubMedBERT. Khác với các mô hình ngôn ngữ được tiền huấn luyện trên văn bản tổng quát, PubMedBERT được tiền huấn luyện từ đầu trên kho ngữ liệu y sinh và sử dụng bộ từ vựng được xây dựng trực tiếp từ miền dữ liệu này. Nhờ đó, mô hình phù hợp hơn với các thuật ngữ chuyên ngành xuất hiện trong thông tin lâm sàng. Chuỗi văn bản mô tả y khoa đầu vào $t$ được chuyển đổi thành vector biểu diễn ngữ nghĩa thông qua các bước biến đổi tuần tự như sau:

\begin{enumerate}
    \item \textbf{Phân tách từ tố (Tokenization):} Văn bản thô $t$ được phân tách thành chuỗi các từ tố (subwords) bằng thuật toán WordPiece dựa trên từ vựng y sinh chuyên biệt gồm $30.000$ từ. Để đảm bảo mô hình bao phủ được các mô tả bệnh lý chi tiết, độ dài tối đa của chuỗi được nới rộng và giới hạn ở ngưỡng $M = 256$ token.
    
    \item \textbf{Khởi tạo chuỗi và chèn các token đặc biệt $[CLS]$ và $[SEP]$:} 
    Để cấu trúc hóa chuỗi đầu vào theo chuẩn BERT, một token phân loại đặc biệt ký hiệu là $t_{\text{CLS}}$ được chèn vào đầu chuỗi, và một token phân tách đặc biệt ký hiệu là $t_{\text{SEP}}$ được bổ sung ở cuối chuỗi để phân định ranh giới ngữ cảnh:
    \begin{equation}
    t_{\text{seq}} = [t_{\text{CLS}}, t_1, t_2, \dots, t_M, t_{\text{SEP}}]
    \end{equation}
    
    Vector nhúng ban đầu $g_0$ sau đó được thiết lập bằng cách cộng phần tử (element-wise summation) giữa các vector nhúng từ tố (token embeddings), nhúng phân đoạn (segment embeddings) và nhúng vị trí (position embeddings) trong không gian ẩn có số chiều $D = 768$:
    \begin{equation}
    g_0 = \text{Embedding}(t_{\text{seq}})
    \end{equation}
    
    \item \textbf{Xử lý qua Transformer Encoder:} Chuỗi biểu diễn tích hợp $g_0$ được truyền qua chồng gồm $L = 12$ tầng Transformer Encoder hai chiều (bidirectional). Tại mỗi tầng, cơ chế tự chú ý đa đầu (Multi-Head Self-Attention) và mạng truyền thẳng (MLP) liên tục cập nhật trạng thái ngữ cảnh của toàn bộ chuỗi thông qua chuẩn hóa trước và kết nối tắt residual.
    
    \item \textbf{Trích xuất đầu ra của bộ mã hóa:} Tại tầng mã hóa cuối cùng ($L=12$), đặc trưng biểu diễn toàn cầu của đoạn văn bản (ký hiệu là $y_t$) được trích xuất bằng cách lấy trạng thái ẩn của token phân loại ở vị trí đầu tiên của chuỗi ($g_L^0$) đi qua một lớp chuẩn hóa LayerNorm cuối cùng:
    \begin{equation}
    y_t = \text{LN}(g_L^0) \in \mathbb{R}^D
    \end{equation}
\end{enumerate}

Kết thúc hai nhánh mã hóa, mỗi mẫu được biểu diễn bởi chuỗi đặc trưng ảnh $\boldsymbol{V}_{i}$, ma trận đặc trưng văn bản $\boldsymbol{H}_{i}^{T}$ và các vectơ toàn cục tương ứng. Mặc dù các biểu diễn này kế thừa tri thức y sinh từ BiomedCLIP, sự khác biệt giữa dữ liệu tiền huấn luyện và dữ liệu X-quang xương đích vẫn đòi hỏi hai bộ mã hóa được thích nghi trước khi thực hiện dung hợp.

\section{Thích nghi tham số hiệu quả bằng LoRA}

Trong quá trình thích nghi các mô hình lớn (như PubMedBERT hay ViT trong BiomedCLIP) cho các tác vụ y sinh hạ nguồn, phương pháp tinh chỉnh toàn bộ tham số đòi hỏi chi phí tính toán và bộ nhớ lưu trữ cực kỳ lớn. Để giải quyết rào cản này, phương pháp Thích nghi tham số hạng thấp (LoRA - Low-Rank Adaptation) được áp dụng. Phương pháp này hoạt động dựa trên giả thuyết rằng sự thay đổi trọng số của các lớp liên kết dày đặc trong quá trình học thực chất nằm trong một không gian có "hạng nội tại" rất thấp. 

Quy trình toán học và dòng chảy dữ liệu của cơ chế LoRA được mô tả cụ thể qua các bước sau:

\begin{enumerate}
    \item \textbf{Đóng băng trọng số nền và thiết lập nhánh song song:}
    Gọi $W_0 \in \mathbb{R}^{d \times k}$ là ma trận trọng số tiền huấn luyện của một lớp tuyến tính bất kỳ trong mạng Transformer. Trong suốt quá trình huấn luyện thích nghi, ma trận $W_0$ này hoàn toàn bị đóng băng và không nhận bất kỳ cập nhật gradient nào. Sự thay đổi trọng số tích lũy $\Delta W$ được biểu diễn gián tiếp thông qua tích của hai ma trận phân rã hạng thấp được đặt song song với lớp gốc:
    \begin{equation}
    W = W_0 + \Delta W = W_0 + B A
    \end{equation}
    Trong đó, $B \in \mathbb{R}^{d \times r}$ và $A \in \mathbb{R}^{r \times k}$ là hai ma trận chứa các tham số học được, với hạng $r \ll \min(d, k)$ là một siêu tham số đại diện cho rank của LoRA.

    \item \textbf{Khởi tạo tham số ban đầu:}
    Để đảm bảo ma trận cập nhật $\Delta W$ bằng $0$ tại thời điểm bắt đầu huấn luyện (tránh gây nhiễu cho tri thức pre-trained gốc), các ma trận phân rã được khởi tạo như sau:
    \begin{itemize}
        \item Ma trận $A$ được khởi tạo ngẫu nhiên theo phân phối Gaussian: $A \sim \mathcal{N}(0, \sigma^2)$.
        \item Ma trận $B$ được khởi tạo bằng $0$.
    \end{itemize}
    Phép nhân ma trận ban đầu mang lại $\Delta W = B \cdot A = 0$, giúp bảo toàn nguyên vẹn đầu ra của mô hình gốc ở vạch xuất phát.

    \item \textbf{Lan truyền tiến tích hợp tham số tỷ lệ:}
    Với một vector đầu vào $x$, phép toán lan truyền tiến (forward pass) qua lớp tuyến tính được điều chỉnh bởi LoRA sẽ cộng trực tiếp kết quả của hai nhánh:
    \begin{equation}
    h = W_0 x + \frac{\alpha}{r} B A x
    \end{equation}
    Trong đó, $\alpha$ là một hằng số tỷ lệ cố định. Việc nhân thêm hệ số điều chỉnh $\frac{\alpha}{r}$ đóng vai trò quan trọng trong việc ổn định hóa quá trình tối ưu hóa, giúp giảm thiểu tối đa việc phải tinh chỉnh lại các siêu tham số huấn luyện khác (như tốc độ học) khi người dùng thay đổi giá trị của hạng $r$.

    \item \textbf{Tích hợp vào kiến trúc Transformer:}
    Mặc dù cấu trúc tuyến tính của LoRA có thể áp dụng cho bất kỳ lớp liên kết dày đặc nào, phương pháp này được ưu tiên áp dụng vào các ma trận trọng số chiếu của cơ chế tự chú ý đa đầu (Multi-Head Self-Attention), cụ thể là các ma trận Query ($W_q$), Key ($W_k$) và Value ($W_v$).
\end{enumerate}
LoRA chỉ điều chỉnh cách từng bộ mã hóa biểu diễn dữ liệu trong miền đích. Bản thân kỹ thuật này chưa thực hiện trao đổi thông tin giữa ảnh và văn bản. Sau khi hai nhánh đã tạo ra các đặc trưng thích nghi, các biểu diễn này được chuyển đến mô-đun dung hợp để mô hình hóa quan hệ giữa dấu hiệu thị giác và ngữ cảnh lâm sàng.

\section{Module dung hợp đa phương thức với chú ý chéo hai chiều}

Sau khi đi qua các bộ mã hóa chuyên biệt, ta thu được các ma trận biểu diễn đặc trưng đầy đủ chứa cả thông tin toàn cục lẫn cục bộ của hai nhánh:
\begin{itemize}
    \item Ma trận đặc trưng hình ảnh $y_v \in \mathbb{R}^{(N+1) \times D}$, trong đó dòng đầu tiên $y_v^0 \in \mathbb{R}^D$ đại diện cho token phân loại $[CLS]$, và phần còn lại $y_v^{1:} \in \mathbb{R}^{N \times D}$ đại diện cho đặc trưng cục bộ của $N$ mảnh ảnh.
    \item Ma trận đặc trưng văn bản $y_t \in \mathbb{R}^{(M+1) \times D}$, trong đó dòng đầu tiên $y_t^0 \in \mathbb{R}^D$ đại diện cho token đặc biệt $[CLS]$, và phần còn lại $y_t^{1:} \in \mathbb{R}^{M \times D}$ đại diện cho đặc trưng cục bộ của $M$ từ tố.
\end{itemize}


\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.5\textwidth]{images/chapter3/fusion.png}
    \caption{Mô-đun dung hợp bằng chú ý chéo hai chiều}
    \label{fig:ch3_fusion}
\end{figure}

Quy trình dung hợp thông tin sử dụng cơ chế chú ý chéo hai chiều (Bidirectional Cross-Attention), trực quan hóa ở hình \ref{fig:ch3_fusion} được thực hiện tuần tự như sau:

\begin{enumerate}
    \item \textbf{Chú ý chéo từ hình ảnh sang văn bản (Image-to-Text Attention):}
    Cơ chế này sử dụng đặc trưng hình ảnh toàn cục $y_v^0$ làm truy vấn (Query) để tìm kiếm và trích xuất ngữ cảnh liên quan từ chuỗi đặc trưng văn bản cục bộ $y_t^{1:}$ đóng vai trò là Keys và Values:
    \begin{equation}
    Q_v = y_v^0 W_q^{(v)}, \quad K_t = y_t^{1:} W_k^{(t)}, \quad V_t = y_t^{1:} W_v^{(t)}
    \end{equation}
    \begin{equation}
    A_{v \to t} = \text{softmax}\left(\frac{Q_v (K_t)^T}{\sqrt{d_k}}\right)
    \end{equation}
    \begin{equation}
    f_{v \to t} = A_{v \to t} V_t \in \mathbb{R}^D
    \end{equation}
    Trong đó, $W_q^{(v)}, W_k^{(t)}, W_v^{(t)} \in \mathbb{R}^{D \times d_k}$ là các ma trận chiếu tuyến tính học được, và $d_k$ là số chiều của không gian chú ý. $A_{v \to t}$ là biểu thị phân phối trọng số thể hiện mức độ tương quan ngữ nghĩa giữa bối cảnh thị giác toàn cục của bức ảnh (Query) và từng từ tố mô tả bệnh lý cục bộ (Keys). Giá trị của từng phần tử chỉ ra mức độ đóng góp ngữ nghĩa của từ tố tương ứng vào việc giải thích hoặc bổ nghĩa cho thực thể trực quan xuất hiện trong ảnh y tế.  Vector $f_{v \to t}$ là biểu diễn văn bản đã được căn chỉnh theo ngữ cảnh hình ảnh.

    \item \textbf{Chú ý chéo từ văn bản sang hình ảnh (Text-to-Image Attention):}
    Tương tự, đặc trưng văn bản toàn cục $y_t^0$ được sử dụng làm Query để rà quét và tập trung vào các vùng đặc trưng không gian trên chuỗi mảnh ảnh cục bộ $y_v^{1:}$:
    \begin{equation}
    Q_t = y_t^0 W_q^{(t)}, \quad K_v = y_v^{1:} W_k^{(v)}, \quad V_v = y_v^{1:} W_v^{(v)}
    \end{equation}
    \begin{equation}
    A_{t \to v} = \text{softmax}\left(\frac{Q_t (K_v)^T}{\sqrt{d_k}}\right)
    \end{equation}
    \begin{equation}
    f_{t \to v} = A_{t \to v} V_v \in \mathbb{R}^D
    \end{equation}
    Vector $f_{t \to v}$ đại diện cho đặc trưng hình ảnh đã được căn chỉnh theo ngữ cảnh của câu mô tả.

    \item \textbf{Chuẩn hóa và Ghép nối đặc trưng (Normalization \& Concatenation):}
    Hai đặc trưng sau khi chú ý chéo được tích hợp lại với các vector toàn cục gốc tương ứng bằng kết nối tắt residual, đi qua lớp chuẩn hóa tầng ($\text{LN}$) rồi tiến hành ghép nối (concatenate) để tạo thành vector dung hợp sơ bộ:
    \begin{equation}
    h_v = \text{LN}(f_{v \to t} + y_v^0) \in \mathbb{R}^D
    \end{equation}
    \begin{equation}
    h_t = \text{LN}(f_{t \to v} + y_t^0) \in \mathbb{R}^D
    \end{equation}
    \begin{equation}
    h_{\text{concat}} = [h_v ; h_t] \in \mathbb{R}^{2D}
    \end{equation}

    \item \textbf{Dung hợp phi tuyến qua MLP để tạo vector tổng hợp cuối cùng:}
    Cuối cùng, vector ghép nối $h_{\text{concat}}$ được đưa qua mạng truyền thẳng ($\text{MLP}$) để thực hiện tương tác phi tuyến tính sâu giữa hai miền dữ liệu, ánh xạ đầu ra về một vector đặc trưng dung hợp tổng hợp duy nhất $y_{\text{fusion}}$ phục vụ tác vụ hạ nguồn:
    \begin{equation}
    y_{\text{fusion}} = \text{MLP}(h_{\text{concat}}) \in \mathbb{R}^D
    \end{equation}
\end{enumerate}

Kết quả của quá trình trên là vectơ đa phương thức $\boldsymbol{y_fusion}$, trong đó thông tin ảnh và văn bản đã được tổng hợp trong cùng một biểu diễn. Vectơ này là đầu vào trực tiếp của bộ phân lớp, do đó quyết định cuối cùng không còn dựa trên từng phương thức riêng lẻ mà dựa trên quan hệ đã học giữa chúng.

\section{Huấn luyện hai giai đoạn trong quá trình ngoại tuyến}

Quy trình ngoại tuyến lần lượt giải quyết hai yêu cầu của hệ thống. Trên nền không gian đã thích nghi, giai đoạn thứ hai học tương tác liên phương thức và tối ưu quyết định phân lớp có xét đến sự mất cân bằng giữa các lớp.

\subsection{Giai đoạn 1: Thích nghi miền bằng khớp ngữ nghĩa}

Giai đoạn thứ nhất được minh họa trong Hình \ref{fig:ch3_phase1}. Trong giai đoạn này, mô hình thực hiện quá trình thích nghi miền nhằm chuyển đổi không gian biểu diễn tổng quát của bộ mã hóa tiền huấn luyện sang miền dữ liệu y khoa chuyên biệt. Để tối ưu hóa hiệu quả và bảo toàn tri thức nền tảng, các tham số gốc của bộ mã hóa hình ảnh và bộ mã hóa văn bản đều được đóng băng, và quá trình huấn luyện thích nghi được thực hiện thông qua cơ chế thích nghi hạng thấp LoRA. 

Trọng tâm của giai đoạn này là giải quyết hiện tượng tiêu cực giả (false negatives) - xảy ra khi các mẫu ảnh và văn bản của các bệnh nhân khác nhau nhưng mang cùng một bệnh lý bị ép phải đẩy ra xa nhau trong không gian nhúng của học tương phản truyền thống. Giai đoạn này sử dụng hàm mất mát khớp ngữ nghĩa (Semantic Matching Loss - SML) kết hợp với ma trận đích mềm (soft target matrix) để dẫn đường cho mô hình.

Quy trình toán học trong một batch huấn luyện kích thước $B$ được mô tả cụ thể như sau:

\begin{enumerate}
    \item \textbf{Xây dựng ma trận đích mềm:}
    Với mỗi cặp thực thể $i$ (hình ảnh) và $j$ (văn bản) trong batch, ta xây dựng một ma trận đích tương đồng mềm $P \in \mathbb{R}^{B \times B}$. Giá trị phân phối mục tiêu $P_{ij}$ đại diện cho mối tương quan y khoa giữa hai mẫu và được định nghĩa theo quy tắc phân cấp ngữ nghĩa:
    \begin{equation}
    P_{ij} = \begin{cases} 
    1.0, & \text{nếu } i = j \quad \text{(cặp khớp)} \\ 
    0.5, & \text{nếu } \text{class}(i) = \text{class}(j) \text{ và } i \neq j \quad \text{(cùng lớp khác mẫu)} \\ 
    0.0, & \text{nếu } \text{class}(i) \neq \text{class}(j) \quad \text{(khác lớp bệnh lý)} 
    \end{cases}
    \end{equation}

    \item \textbf{Chuẩn hóa phân phối mục tiêu:}
    Để biến đổi ma trận đích mềm $P$ thành phân phối xác suất hợp lệ, ta áp dụng phép chuẩn hóa dòng (từ ảnh sang văn bản) và chuẩn hóa cột (từ văn bản sang ảnh) thông qua hàm Softmax:
    \begin{equation}
    y_{ij}^{v \to t} = \frac{\exp(P_{ij})}{\sum_{k=1}^{B} \exp(P_{ik})}
    \end{equation}
    \begin{equation}
    y_{ji}^{t \to v} = \frac{\exp(P_{ji})}{\sum_{k=1}^{B} \exp(P_{ki})}
    \end{equation}

    \item \textbf{Tính toán tương đồng dự đoán:}
    Gọi $I_i$ và $T_j$ lần lượt là các vector nhúng chuẩn hóa $L_2$ của hình ảnh thứ $i$ và văn bản thứ $j$ trong không gian chung. Lượng logits tương đồng dự đoán scaled cosine similarity $\hat{s}_{ij}$ được tính toán bằng:
    \begin{equation}
    \hat{s}_{ij} = \frac{I_i \cdot T_j}{\tau}
    \end{equation}
    Trong đó, $\tau$ là tham số nhiệt độ học được. Phân phối xác suất dự đoán tương ứng được xác định bởi:
    \begin{equation}
    \hat{y}_{ij}^{v \to t} = \frac{\exp(\hat{s}_{ij})}{\sum_{k=1}^{B} \exp(\hat{s}_{ik})}, \quad 
    \hat{y}_{ji}^{t \to v} = \frac{\exp(\hat{s}_{ji})}{\sum_{k=1}^{B} \exp(\hat{s}_{ki})}
    \end{equation}

    \item \textbf{Hàm mất mát khớp ngữ nghĩa (Semantic Matching Loss):}
    Hàm mất mát SML được định nghĩa là trung bình cộng của sai số entropy chéo hai chiều đối xứng giữa phân phối đích mềm và phân phối dự đoán của mô hình thích nghi LoRA:
    \begin{equation}
    \mathcal{L}_{\text{SML}} = \frac{\mathcal{L}_{v \to t} + \mathcal{L}_{t \to v}}{2}
    \end{equation}
    Trong đó:
    \begin{equation}
    \mathcal{L}_{v \to t} = -\frac{1}{B} \sum_{i=1}^{B} \sum_{j=1}^{B} y_{ij}^{v \to t} \log \hat{y}_{ij}^{v \to t}
    \end{equation}
    \begin{equation}
    \mathcal{L}_{t \to v} = -\frac{1}{B} \sum_{j=1}^{B} \sum_{i=1}^{B} y_{ji}^{t \to v} \log \hat{y}_{ji}^{t \to v}
    \end{equation}
\end{enumerate}

Trong giai đoạn thứ hai (Phase 2), mục tiêu cốt lõi của mạng là tiến hành dung hợp sâu sắc các thông tin đa phương thức từ hai nhánh ảnh-chữ, sau đó đưa qua một bộ phân loại tuyến tính (Linear Classifier) đơn giản để dự đoán trực tiếp các lớp bệnh lý hạ nguồn. 

Quy trình lan truyền tiến, tính toán hàm mất mát và cập nhật trọng số trong giai đoạn này được thực hiện tuần tự theo các bước sau:

\begin{enumerate}
    \item \textbf{Khởi tạo đầu vào bộ phân loại:}
    Gọi $\mathbf{z} = y_{\text{fusion}} \in \mathbb{R}^D$ là vector đặc trưng dung hợp tổng hợp thu được từ đầu ra của module chú ý chéo hai chiều (với số chiều ẩn $D = 768$). Bộ phân loại hạ nguồn được thiết lập để dự đoán $C = 22$ lớp bệnh lý y khoa mục tiêu.

    \item \textbf{Phép chiếu tuyến tính và tính xác suất dự đoán (Forward Pass):}
    Vector đặc trưng dung hợp $\mathbf{z}$ được đưa trực tiếp qua một lớp tuyến tính đơn giản (Linear Layer) để tính toán ma trận logits điểm số $\mathbf{s} \in \mathbb{R}^{C}$:
    \begin{equation}
    \mathbf{s} = W_{\text{clf}} \mathbf{z} + b_{\text{clf}}
    \end{equation}
    Trong đó, $W_{\text{clf}} \in \mathbb{R}^{C \times D}$ là ma trận trọng số khả huấn của bộ phân loại, và $b_{\text{clf}} \in \mathbb{R}^{C}$ là vector sai số (bias).
    
    Điểm logits thô $s_c$ của từng lớp $c \in [1, C]$ sau đó được đưa qua hàm kích hoạt Softmax để chuyển đổi thành phân phối xác suất dự đoán $\hat{p}_c \in [1]$ tương ứng:
    \begin{equation}
    \hat{p}_c = \text{Softmax}(s_c) = \frac{\exp(s_c)}{\sum_{j=1}^{C} \exp(s_j)}
    \end{equation}

    \item \textbf{Hàm mất mát Entropy chéo có trọng số (Weighted Cross-Entropy Loss):}
    Để đối phó với hiện tượng mất cân bằng mẫu nghiêm trọng giữa 22 lớp bệnh lý trong tập dữ liệu y khoa, mô hình áp dụng hàm mất mát Weighted Cross-Entropy ($\mathcal{L}_{\text{WCE}}$). Với nhãn thực tế dạng mã hóa một nóng (one-hot) ký hiệu là $y \in \{0, 1\}^{C}$, hàm mất mát được định nghĩa như sau:
    \begin{equation}
    \mathcal{L}_{\text{WCE}} = - \sum_{c=1}^{C} w_c y_c \log(\hat{p}_c)
    \end{equation}
    Trong đó, $w_c$ là hệ số trọng số phạt được gán cho lớp thứ $c$ (tỉ lệ nghịch với tần suất xuất hiện của lớp đó trong tập huấn luyện).

    \item \textbf{Lan truyền ngược sai số gradient (Backpropagation):}
    Để tối ưu hóa toàn bộ mô hình, mạng tính toán đạo hàm riêng của hàm lỗi $\mathcal{L}_{\text{WCE}}$ đối với các điểm logit đầu ra $s_i$ ($i = 1, \dots, C$):
    \begin{equation}
    \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial s_i} = \left( \sum_{c=1}^{C} w_c y_c \right) \hat{p}_i - w_i y_i
    \end{equation}
    Giả sử mẫu dữ liệu thuộc về lớp thực tế $c^*$ (tức $y_{c^*} = 1$ và $y_{i \neq c^*} = 0$), phương trình trên được thu gọn tối giản về dạng:
    \begin{equation}
    \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial s_i} = w_{c^*} (\hat{p}_i - y_i)
    \end{equation}
    
    Gradient của hàm lỗi đối với các tham số trực tiếp của lớp phân loại tuyến tính ($W_{\text{clf}}$ và $b_{\text{clf}}$) được xác định bằng:
    \begin{equation}
    \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial W_{\text{clf}}} = \left( \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial \mathbf{s}} \right) \mathbf{z}^T, \quad \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial b_{\text{clf}}} = \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial \mathbf{s}}
    \end{equation}

    Để tiếp tục cập nhật các khối bên dưới, gradient sai số được truyền ngược về vector dung hợp đầu vào $\mathbf{z}$:
    \begin{equation}
    \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial \mathbf{z}} = W_{\text{clf}}^T \left( \frac{\partial \mathcal{L}_{\text{WCE}}}{\partial \mathbf{s}} \right)
    \end{equation}

    \item \textbf{Quy trình cập nhật trọng số hệ thống:}
    Gradient $\frac{\partial \mathcal{L}_{\text{WCE}}}{\partial \mathbf{z}}$ tiếp tục lan truyền ngược qua khối MLP, các lớp chuẩn hóa tầng LN, các ma trận chiếu của module chú ý chéo hai chiều, và đi sâu xuống các ma trận thích ứng LoRA ($A$ và $B$) của cả hai nhánh bộ mã hóa ảnh và chữ. 
    
    Gọi $\Theta_2$ là tập hợp tất cả các tham số có quyền cập nhật trong giai đoạn này. Tại mỗi bước lặp $t$, thuật toán tối ưu hóa AdamW thực hiện cập nhật đồng thời các tham số theo công thức:
    \begin{equation}
    \Theta_2^{(t+1)} = \Theta_2^{(t)} - \eta \cdot \text{AdamW}\left(\nabla_{\Theta_2} \mathcal{L}_{\text{WCE}}^{(t)}\right)
    \end{equation}
    Trong đó, $\eta$ là tốc độ học được thiết lập theo lịch trình suy giảm tuyến tính.
\end{enumerate}

\section{Phát hiện dữ liệu lệch phân phối bằng thuật toán k-NN đa phương thức}

Sau khi huấn luyện thích nghi mô hình, các bộ mã hóa hình ảnh và văn bản có khả năng trích xuất các đặc trưng biểu diễn ngữ nghĩa rất mạnh mẽ. Tuy nhiên, khi triển khai thực tế, mô hình có thể gặp phải các mẫu dữ liệu lệch phân phối (Out-of-Distribution - OOD) nằm ngoài tập huấn luyện gốc. Để nhận diện và ngăn chặn các mẫu dữ liệu này, hệ thống áp dụng phương pháp phát hiện OOD phi tham số dựa trên thuật toán $k$-láng giềng gần nhất ($k$-NN) hoạt động đồng thời trên cả hai nhánh ảnh và chữ.

Thay vì sử dụng vector đặc trưng dung hợp cuối cùng $y_{\text{fusion}}$ trích xuất từ module chú ý chéo ở Giai đoạn 2, phương pháp đề xuất lựa chọn trích xuất song song hai vector đặc trưng độc lập $y_v$ và $y_t$ sát với đầu ra của bộ mã hóa gốc. Vector dung hợp $y_{\text{fusion}}$ được tối ưu hóa trực tiếp bằng hàm lỗi Weighted Cross-Entropy nhằm phân tách rõ rệt giữa các lớp bệnh lý mục tiêu. Quá trình này sẽ ép mô hình thực hiện nén dữ liệu và loại bỏ các đặc trưng không liên quan đến tác vụ, vô tình triệt tiêu các thông tin phân phối nhạy cảm cần thiết để nhận diện dữ liệu lạ. Việc trích xuất đặc trưng sát bộ mã hóa gốc (tương đương với đặc trưng tầng áp chót - penultimate layer) giúp bảo toàn được nhiều thông tin phân phối phong phú và nguyên bản nhất.

Quy trình tính toán điểm số OOD và thiết lập ngưỡng quyết định được thực hiện tuần tự qua các bước sau:

\begin{enumerate}
    \item \textbf{Chuẩn hóa đặc trưng đầu vào:}
    Gọi $y_v \in \mathbb{R}^D$ và $y_t \in \mathbb{R}^D$ (với $D = 768$) lần lượt là các vector đặc trưng biểu diễn hình ảnh và văn bản thu được từ tầng cuối cùng của các bộ mã hóa tương ứng. Trước khi tính toán khoảng cách, hai vector này bắt buộc phải được chuẩn hóa chuẩn $L_2$ để đưa về mặt cầu đơn vị:
    \begin{equation}
    \tilde{y}_v = \frac{y_v}{\|y_v\|_2}, \quad \tilde{y}_t = \frac{y_t}{\|y_t\|_2}
    \end{equation}
    Việc chuẩn hóa này đóng vai trò quyết định, giúp loại bỏ sự ảnh hưởng của độ dài vector (norm) vốn có xu hướng phân phối rất khác biệt giữa dữ liệu trong phân phối (ID) và dữ liệu ngoại lai (OOD), đồng thời đưa phép đo khoảng cách Euclid về tương đương với độ tương đồng Cosine.

    \item \textbf{Thiết lập cơ sở dữ liệu tham chiếu trong phân phối (ID):}
    Từ tập dữ liệu huấn luyện trong phân phối (In-Distribution), ta trích xuất và lưu trữ sẵn hai tập hợp vector đặc trưng đã chuẩn hóa làm cơ sở dữ liệu đối sánh:
    \begin{equation}
    \mathcal{D}_v = \{z_{v, 1}, z_{v, 2}, \dots, z_{v, N_{\text{train}}}\}, \quad \mathcal{D}_t = \{z_{t, 1}, z_{t, 2}, \dots, z_{t, N_{\text{train}}}\}
    \end{equation}

    \item \textbf{Tính toán khoảng cách $k$-NN cho từng nhánh độc lập:}
    Với một mẫu thử nghiệm đa phương thức có cặp đặc trưng đầu vào là $(\tilde{y}_v, \tilde{y}_t)$, ta lần lượt tính toán khoảng cách Euclid đến tất cả các mẫu trong tập tham chiếu tương ứng, sau đó sắp xếp theo thứ tự tăng dần. Khoảng cách đến láng giềng thứ $k$ (ký hiệu là $d_v$ cho ảnh và $d_t$ cho văn bản) được xác định bởi:
    \begin{equation}
    d_v(\tilde{y}_v) = \|\tilde{y}_v - z_{v}^{(k)}\|_2
    \end{equation}
    \begin{equation}
    d_t(\tilde{y}_t) = \|\tilde{y}_t - z_{t}^{(k)}\|_2
    \end{equation}
    Trong đó, $z_{v}^{(k)}$ và $z_{t}^{(k)}$ lần lượt là các vector láng giềng gần nhất thứ $k$ của đặc trưng ảnh và đặc trưng văn bản thử nghiệm trong không gian biểu diễn tương ứng.

    \item \textbf{Dung hợp điểm số OOD bằng toán tử cực đại (Max Fusion):}
    Để đảm bảo tính nhạy bén tối đa của hệ thống trước sự lệch phân phối ngữ nghĩa xảy ra ở bất kỳ phương thức nào (hoặc chỉ lệch ở ảnh, hoặc chỉ lệch ở chữ, hoặc lệch ở cả hai), điểm số khoảng cách OOD tổng hợp ($d_{\text{OOD}}$) được định nghĩa bằng giá trị cực đại giữa hai khoảng cách độc lập:
    \begin{equation}
    d_{\text{OOD}}(\tilde{y}_v, \tilde{y}_t) = \max \left( d_v(\tilde{y}_v), \, d_t(\tilde{y}_t) \right)
    \end{equation}

    \item \textbf{Xác định ngưỡng quyết định dựa trên tập kiểm định:}
    Ngưỡng phát hiện OOD (ký hiệu là $\lambda$) được thiết lập hoàn toàn dựa trên dữ liệu trong phân phối (ID) mà không cần bất kỳ thông tin nào của dữ liệu ngoại lai chưa biết. Cụ thể, $\lambda$ được chọn là giá trị phân vị thứ 95 (95\% quantile) của tập hợp các điểm số kết hợp $d_{\text{OOD}}$ tính toán trên tập kiểm định trong phân phối (Validation ID set - $\mathcal{V}_{\text{ID}}$):
    \begin{equation}
    P\left( d_{\text{OOD}}(\tilde{y}_v, \tilde{y}_t) \le \lambda \;\middle|\; (\tilde{y}_v, \tilde{y}_t) \in \mathcal{V}_{\text{ID}} \right) = 0.95
    \end{equation}
    Ngưỡng này đảm bảo về mặt thống kê rằng hệ thống sẽ giữ lại chính xác 95\% dữ liệu ID thực tế (tương ứng với tỷ lệ True Positive Rate - TPR95 đạt mức 95\%).

    \item \textbf{Hàm quyết định phân loại quyết định:}
    Tại pha thử nghiệm, mô hình đưa ra quyết định phân loại nhị phân cuối cùng cho mẫu dựa trên việc so sánh điểm số $d_{\text{OOD}}$ với ngưỡng $\lambda$:
    \begin{equation}
    G(\tilde{y}_v, \tilde{y}_t) = \begin{cases} 
    \text{ID (Trong phân phối)}, & \text{nếu } d_{\text{OOD}}(\tilde{y}_v, \tilde{y}_t) \le \lambda \\ 
    \text{OOD (Lệch phân phối)}, & \text{nếu } d_{\text{OOD}}(\tilde{y}_v, \tilde{y}_t) > \lambda 
    \end{cases}
    \end{equation}
\end{enumerate}

\begin{figure}[htbp]
    \centering
    \includegraphics[width=\textwidth]{images/chapter3/online.png}
    \caption{Quy trình suy luận trực tuyến và phát hiện mẫu ngoài phân phối}

Tại pha suy luận (inference), khi nhận được một mẫu dữ liệu bệnh nhân mới bao gồm cặp hình ảnh y tế đầu vào $v$ và văn bản mô tả lâm sàng $t$, hệ thống thực hiện một quy trình xử lý khép kín, an toàn và tối ưu hóa thời gian phản hồi. 

Để triệt tiêu hoàn toàn độ trễ tính toán phát sinh từ các nhánh thích ứng bổ trợ trong pha suy luận thực tế, các ma trận trọng số LoRA học được ($A$ và $B$) được cộng tích hợp trực tiếp vào ma trận trọng số gốc của các lớp tuyến tính tương ứng trong bộ mã hóa ảnh và văn bản trước khi triển khai:
\begin{equation}
W_{\text{inference}} = W_0 + \frac{\alpha}{r} B A
\end{equation}
Nhờ phép cộng gộp này, mô hình suy luận hoạt động với cấu trúc và số lượng tham số tương đương mô hình pre-trained gốc nhưng đã mang đầy đủ tri thức thích nghi miền. Dòng chảy dữ liệu và quyết định của quy trình suy luận diễn ra tuần tự qua các bước sau:

\begin{enumerate}
    \item \textbf{Trích xuất đặc trưng song song (Parallel Feature Extraction):}
    Cặp đầu vào $(v, t)$ được đưa vào hai bộ mã hóa thích ứng để trích xuất các ma trận đặc trưng biểu diễn ngữ nghĩa:
    \begin{itemize}
        \item Bộ mã hóa hình ảnh trích xuất ma trận đặc trưng đầy đủ $y_v \in \mathbb{R}^{(N+1) \times D}$, được chia thành vector toàn cục ở dòng đầu tiên $y_v^0 \in \mathbb{R}^D$ (token $[CLS]$) và ma trận đặc trưng cục bộ của các mảnh ảnh ở các dòng còn lại $y_v^{1:} \in \mathbb{R}^{N \times D}$.
        \item Bộ mã hóa văn bản trích xuất ma trận đặc trưng đầy đủ $y_t \in \mathbb{R}^{(M+1) \times D}$, được chia thành vector toàn cục ở dòng đầu tiên $y_t^0 \in \mathbb{R}^D$ (token $[CLS]$) và ma trận đặc trưng cục bộ của các từ tố ở các dòng còn lại $y_t^{1:} \in \mathbb{R}^{M \times D}$.
    \end{itemize}

    \item \textbf{Kiểm duyệt an toàn OOD bằng k-NN đa phương thức (Safe OOD Screening):}
    Trước khi đưa dữ liệu vào module dung hợp, các vector đặc trưng toàn cục được đưa qua cổng kiểm duyệt an toàn để nhận diện dữ liệu lệch phân phối (OOD):
    \begin{itemize}
        \item Tiến hành chuẩn hóa chuẩn $L_2$ trên các vector toàn cục để đưa về mặt cầu đơn vị:
        \begin{equation}
        \tilde{y}_v = \frac{y_v^0}{\|y_v^0\|_2}, \quad \tilde{y}_t = \frac{y_t^0}{\|y_t^0\|_2}
        \end{equation}
        \item Tính toán khoảng cách Euclid $d_v$ và $d_t$ từ các vector đã chuẩn hóa đến láng giềng thứ $k$ gần nhất trong hai tập cơ sở dữ liệu tham chiếu trong phân phối ($\mathcal{D}_v$ và $\mathcal{D}_t$).
        \item Dung hợp điểm số OOD bằng toán tử cực đại để phát hiện nhạy bén sự bất thường ở bất kỳ phương thức nào:
        \begin{equation}
        d_{\text{OOD}} = \max \left( d_v, \, d_t \right)
        \end{equation}
        \item Đối sánh điểm số OOD với ngưỡng quyết định $\lambda$ (thiết lập từ phân vị 95\% trên tập kiểm định ID):
        \begin{itemize}
            \item \textbf{Trường hợp OOD ($d_{\text{OOD}} > \lambda$):} Mẫu thử nghiệm bị xác định là dữ liệu lạ hoặc bị lỗi nhiễu nghiêm trọng. Hệ thống ngay lập tức \textbf{dừng quy trình suy luận}, từ chối đưa ra chẩn đoán tự động và phát cảnh báo yêu cầu bác sĩ chuyên khoa can thiệp kiểm tra thủ công.
            \item \textbf{Trường hợp ID ($d_{\text{OOD}} \le \lambda$):} Mẫu thử nghiệm được xác nhận an toàn (thuộc miền phân phối bệnh lý đã học) và được phép chuyển tiếp sang bước chẩn đoán tiếp theo.
        \end{itemize}
    \end{itemize}

    \item \textbf{Dung hợp ngữ cảnh bằng chú ý chéo hai chiều (Bidirectional Cross-Attention Fusion):}
    Với dữ liệu an toàn, mô hình thực hiện dung hợp sâu sắc thông tin đa phương thức:
    \begin{itemize}
        \item Nhánh ảnh-sang-chữ dùng Query $y_v^0$ rà quét chuỗi cục bộ $y_t^{1:}$ để tạo ra đặc trưng chữ căn chỉnh bối cảnh ảnh $f_{v \to t} \in \mathbb{R}^D$.
        \item Nhánh chữ-sang-ảnh dùng Query $y_t^0$ rà quét chuỗi cục bộ $y_v^{1:}$ để tạo ra đặc trưng ảnh căn chỉnh bối cảnh chữ $f_{t \to v} \in \mathbb{R}^D$.
        \item Thực hiện chuẩn hóa LN kết hợp kết nối tắt và ghép nối (concatenation) đặc trưng:
        \begin{equation}
        h_v = \text{LN}(f_{v \to t} + y_v^0), \quad h_t = \text{LN}(f_{t \to v} + y_t^0), \quad h_{\text{concat}} = [h_v ; h_t] \in \mathbb{R}^{2D}
        \end{equation}
        \item Đưa qua mạng truyền thẳng ($\text{MLP}$) để tạo vector biểu diễn dung hợp tổng hợp cuối cùng:
        \begin{equation}
        \mathbf{z} = y_{\text{fusion}} = \text{MLP}(h_{\text{concat}}) \in \mathbb{R}^D
        \end{equation}
    \end{itemize}

    \item \textbf{Phân lớp chẩn đoán bệnh lý (Diagnostic Classification):}
    Vector dung hợp tối ưu $\mathbf{z} \in \mathbb{R}^{768}$ được đưa qua bộ phân loại tuyến tính đơn giản cuối cùng để đưa ra quyết định lâm sàng:
    \begin{itemize}
        \item Tính vector logits điểm số $\mathbf{s}$ cho 22 lớp bệnh lý hạ nguồn:
        \begin{equation}
        \mathbf{s} = W_{\text{clf}} \mathbf{z} + b_{\text{clf}} \in \mathbb{R}^{22}
        \end{equation}
        \item Áp dụng hàm kích hoạt Softmax để thu được phân phối xác suất tin cậy của mô hình trên từng lớp $c \in [1, 2]$:
        \begin{equation}
        \hat{p}_c = \frac{\exp(s_c)}{\sum_{j=1}^{22} \exp(s_j)}
        \end{equation}
        \item Hệ thống xuất ra lớp bệnh lý có xác suất dự đoán $\hat{p}_c$ cao nhất (hoặc danh sách các bệnh lý kèm tỷ lệ phần trăm tin cậy tương ứng) làm chẩn đoán cuối cùng hỗ trợ bác sĩ ra quyết định điều trị.
    \end{itemize}
\end{enumerate}

Bên cạnh hai thành phần dự đoán trên, đối với mẫu không bị cảnh báo ngoài phân phối, quá trình suy luận còn có thể cung cấp thông tin hỗ trợ giải thích bằng phương pháp tích phân gradient \cite{Sundararajan2017IntegratedGradients}. Thành phần này được thực hiện như một lượt tính riêng khi cần giải thích. Các tham số mô hình vẫn được cố định, còn đạo hàm chỉ được tính đối với đầu vào hoặc biểu diễn cần khảo sát và không được dùng để cập nhật mô hình.

Với mỗi mẫu, lớp đích được chọn là lớp do mô hình dự đoán $\widehat{y}$, kể cả khi dự đoán này khác nhãn tham chiếu. Gọi $F_{\widehat{y}}(\boldsymbol{u})$ là giá trị logit của lớp đích tại đầu vào hoặc biểu diễn $\boldsymbol{u}$, và $\boldsymbol{u}^{0}$ là mốc tham chiếu tương ứng. Quy đóng của thành phần thứ $j$ được xác định bởi:

\begin{equation}
\operatorname{IG}_{j}(\boldsymbol{u})
=
\left(u_{j}-u_{j}^{0}\right)
\int_{0}^{1}
\frac{\partial F_{\widehat{y}}
\!\left(\boldsymbol{u}^{0}+\beta(\boldsymbol{u}-\boldsymbol{u}^{0})\right)}
{\partial u_{j}}
\,\mathrm{d}\beta,
\label{eq:ch3_integrated_gradients}
\end{equation}

trong đó $\beta\in[0,1]$ xác định một điểm trên đường thẳng từ mốc tham chiếu đến đầu vào. Trong phần đánh giá thực nghiệm, tích phân được xấp xỉ bằng quy tắc hình thang tại 24 điểm cách đều. Đối với ảnh toàn cục, mốc tham chiếu là tenxơ bằng không trong không gian đầu vào đã chuẩn hóa. Đối với các đơn vị ảnh cục bộ, mốc tham chiếu là biểu diễn ảnh toàn cục được lặp lại theo số đơn vị. Đối với bệnh sử, các từ nội dung được nội suy từ biểu diễn của ký hiệu đệm, trong khi các ký hiệu đặc biệt được giữ nguyên.

Để thu được mức liên quan của từng điểm ảnh, đơn vị ảnh cục bộ hoặc từ, các quy đóng theo chiều biểu diễn được cộng lại và chỉ giữ phần dương. Nếu tổng phần dương gần bằng không, tổng trị tuyệt đối được sử dụng làm phương án dự phòng. Bản đồ và điểm liên quan thu được phản ánh mức đóng góp của từng thành phần đối với logit của lớp dự đoán trong lân cận đường nội suy, do đó được sử dụng như thông tin hỗ trợ diễn giải thay vì một kết luận nhân quả.


Khi tiếp nhận một ca mới, ảnh X-quang và thông tin lâm sàng được xử lý bằng đúng các phép biến đổi đã sử dụng trong huấn luyện. Hai bộ mã hóa tạo đặc trưng ảnh và văn bản, mô-đun chú ý chéo tạo vectơ tổng hợp, sau đó bộ phân lớp tính xác suất cho từng lớp. Toàn bộ tham số LoRA, mô-đun dung hợp và các nguyên mẫu lớp đều được cố định. Bước phân lớp và tính điểm ngoài phân phối không tính đạo hàm, đồng thời không cập nhật tâm lớp từ mẫu mới. Kết quả phân loại cho biết lớp nào phù hợp nhất trong số các lớp đã biết, nhưng chưa cho biết mẫu có thực sự thuộc phân bố huấn luyện hay không.

Để phát hiện mẫu ngoài phân phối (OOD), việc chỉ dựa vào không gian biểu diễn sau khi dung hợp ($\boldsymbol{z}_{\mathrm{fused}}$) có thể gặp hạn chế do mô-đun chú ý chéo có xu hướng ép các đặc trưng dị biệt vào cấu trúc không gian lớp nội tại để tối ưu phân loại, làm suy giảm độ nhạy phát hiện bất thường ở từng phương thức riêng lẻ. Vì vậy, hệ thống xây dựng cơ chế \textbf{Multi-Modal Modality-Decoupled Ensemble OOD Detection} bằng cách tách rời hai phương thức, áp dụng phương pháp khoảng cách $k$-láng giềng gần nhất ($k\mathrm{NN}$) trên từng không gian biểu diễn đơn phương thức, chuẩn hóa $Z$-score trên tập xác thực ID và hợp nhất bằng toán tử $\max$:

\begin{enumerate}
    \item \textbf{Khoảng cách $k\mathrm{NN}$ đơn phương thức tách rời:} Gọi $\mathcal{D}_{\mathrm{ID}}^{\mathrm{train}}$ là tập biểu diễn tham chiếu trong phân phối đã được chuẩn hóa $L_2$. Với mỗi mẫu kiểm thử có biểu diễn ảnh $\boldsymbol{z}_{\mathrm{img}}$ và biểu diễn văn bản $\boldsymbol{z}_{\mathrm{txt}}$, điểm bất thường đơn phương thức được tính bằng khoảng cách Euclid tới láng giềng thứ $k$ gần nhất \cite{Sun2022KNN}:
    \begin{equation}
    s_{\mathrm{vis}}(\boldsymbol{z}_{\mathrm{img}}) = d_{k\mathrm{NN}}\!\left(\boldsymbol{z}_{\mathrm{img}},\,\mathcal{D}_{\mathrm{ID},\,\mathrm{img}}^{\mathrm{train}}\right),
    \label{eq:ch3_ood_knn_vis}
    \end{equation}
    \begin{equation}
    s_{\mathrm{txt}}(\boldsymbol{z}_{\mathrm{txt}}) = d_{k\mathrm{NN}}\!\left(\boldsymbol{z}_{\mathrm{txt}},\,\mathcal{D}_{\mathrm{ID},\,\mathrm{txt}}^{\mathrm{train}}\right).
    \label{eq:ch3_ood_knn_txt}
    \end{equation}

    \item \textbf{Chuẩn hóa $Z$-score trên tập ID Validation:} Do thang đo khoảng cách trên không gian ảnh và không gian văn bản có thể khác biệt, hệ thống ước lượng giá trị kỳ vọng và độ lệch chuẩn của từng phương thức trên tập xác thực $\mathcal{D}_{\mathrm{ID}}^{\mathrm{val}}$ ($\mu_{\mathrm{vis}}^{\mathrm{val}}, \sigma_{\mathrm{vis}}^{\mathrm{val}}$ và $\mu_{\mathrm{txt}}^{\mathrm{val}}, \sigma_{\mathrm{txt}}^{\mathrm{val}}$) để đưa các điểm số về phân phối chuẩn hóa không thứ nguyên:
    \begin{equation}
    \widetilde{s}_{\mathrm{vis}} = \frac{s_{\mathrm{vis}}(\boldsymbol{z}_{\mathrm{img}}) - \mu_{\mathrm{vis}}^{\mathrm{val}}}{\sigma_{\mathrm{vis}}^{\mathrm{val}} + \epsilon}, \qquad \widetilde{s}_{\mathrm{txt}} = \frac{s_{\mathrm{txt}}(\boldsymbol{z}_{\mathrm{txt}}) - \mu_{\mathrm{txt}}^{\mathrm{val}}}{\sigma_{\mathrm{txt}}^{\mathrm{val}} + \epsilon}.
    \label{eq:ch3_ood_zscore}
    \end{equation}

    \item \textbf{Hợp nhất bằng toán tử $\max$ (Max-pooling Fusion):} Điểm ngoài phân phối tổng hợp được xác định theo nguyên lý phòng vệ nghiêm ngặt (worst-case anomaly):
    \begin{equation}
    s_{\mathrm{OOD}}(\boldsymbol{x}) = \max\left(\widetilde{s}_{\mathrm{vis}},\,\widetilde{s}_{\mathrm{txt}}\right).
    \label{eq:ch3_ood_max_fusion}
    \end{equation}
    Theo cơ chế này, chỉ cần một trong hai phương thức (ảnh hoặc văn bản lâm sàng) có mức độ bất thường vượt trội so với phân phối huấn luyện, toàn bộ ca bệnh sẽ lập tức được nhận diện và cảnh báo là dữ liệu ngoài phân phối.
\end{enumerate}

Ngưỡng cảnh báo $\tau_{\mathrm{alert}}$ được hiệu chỉnh trên tập xác thực $\mathcal{D}_{\mathrm{ID}}^{\mathrm{val}}$ ở mức tỷ lệ cảnh báo giả mục tiêu $\alpha = 0{,}05$:
\begin{equation}
\tau_{\mathrm{alert}} = Q_{1-\alpha}^{+}\!\left(\left\{s_{\mathrm{OOD}}(\boldsymbol{x}_{i}) \mid \boldsymbol{x}_{i}\in\mathcal{D}_{\mathrm{ID}}^{\mathrm{val}}\right\}\right), \qquad \alpha=0{,}05,
\label{eq:ch3_ood_threshold}
\end{equation}
trong đó $Q_{q}^{+}$ là phân vị thực nghiệm phía trên ở mức $q$, nhằm bảo đảm giữ lại ít nhất $95\%$ mẫu trong phân phối. Quy tắc đưa ra cảnh báo OOD độc lập với kết quả phân lớp:
\begin{equation}
\begin{aligned}
\widehat{y} &= \operatorname*{arg\,max}_{k\in\mathcal{Y}} p(y=k\mid\boldsymbol{x}), \\[6pt]
\omega_{\mathrm{OOD}} &= \begin{cases}
1, & \text{nếu } s_{\mathrm{OOD}}(\boldsymbol{x}) > \tau_{\mathrm{alert}}, \\[4pt]
0, & \text{ngược lại}.
\end{cases}
\end{aligned}
\label{eq:ch3_ood_decision}
\end{equation}

Mô-đun phát hiện ngoài phân phối không thay đổi dự đoán nội tại của XBone-Net mà cung cấp thêm tín hiệu cảnh báo $\omega_{\mathrm{OOD}}$ khi mẫu đầu vào có dấu hiệu bất thường về thị giác hoặc bất thường về nội dung văn bản lâm sàng.
